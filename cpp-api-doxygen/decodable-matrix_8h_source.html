273 int32_t NumFramesReady() const override {
274 return loglikes_.rows() + frame_offset_;
275 }
276
277 bool IsLastFrame(int32_t frame) const override {
278 if (frame >= NumFramesReady()) {
279 throw std::runtime_error(
280 "DecodableMatrixMappedOffset::IsLastFrame: frame out of range");
281 }
282 return frame == NumFramesReady() - 1 && input_is_finished_;
283 }
284
285 float LogLikelihood(int32_t frame, int32_t tid) override {
286 if (tid < 1 || tid >= static_cast<int32_t>(tid_to_pdf_.size())) {
287 throw std::runtime_error(
288 "DecodableMatrixMappedOffset::LogLikelihood: tid out of range");
289 }
290 if (frame < frame_offset_ || frame >= NumFramesReady()) {
291 throw std::runtime_error(
292 "DecodableMatrixMappedOffset::LogLikelihood: frame out of range");
293 }
294 int32_t pdf_id = tid_to_pdf_[tid];
295 return loglikes_(frame - frame_offset_, pdf_id);
296 }
297
298 // Note: these indices are 1-based.
299 int32_t NumIndices() const override {
300 return trans_model_.NumTransitionIds();
301 }
302
303 // nothing special to do in destructor.
304 ~DecodableMatrixMappedOffset() override = default;
305
306 private:
308 const TransitionInformation &trans_model_;
310 const std::vector<int32_t> &tid_to_pdf_;
312 FloatMatrix loglikes_;
314 int32_t frame_offset_ = 0;
316 bool input_is_finished_ = false;
317
318 public:
319 DecodableMatrixMappedOffset(const DecodableMatrixMappedOffset &) = delete;
320 DecodableMatrixMappedOffset &operator=(const DecodableMatrixMappedOffset &) =
321 delete;

329class DecodableMatrixScaled : public DecodableInterface {
330 public:
331 DecodableMatrixScaled(const FloatMatrix &likes, float scale)
332 : likes_(likes), scale_(scale) {}
333
334 int32_t NumFramesReady() const override { return likes_.rows(); }
335
336 bool IsLastFrame(int32_t frame) const override {
337 if (frame >= NumFramesReady()) {
338 throw std::runtime_error(
339 "DecodableMatrixScaled::IsLastFrame: frame out of range");
340 }
341 return frame == NumFramesReady() - 1;
342 }
343
344 // Note, frames are numbered from zero.
345 float LogLikelihood(int32_t frame, int32_t index) override {
346 if (index > likes_.cols() || index <= 0 || frame < 0 ||
347 frame >= likes_.rows()) {
348 std::ostringstream os;
349 os << "DecodableMatrixScaled: invalid (frame, index - 1) = (" << frame
350 << ", " << index - 1 << ") for matrix of size " << likes_.rows()
351 << " x " << likes_.cols();
352 throw std::runtime_error(os.str());
353 }
354 return scale_ * likes_(frame, index - 1);
355 }
356
357 // Indices are one-based! This is for compatibility with OpenFst.
358 int32_t NumIndices() const override { return likes_.cols(); }
359
360 private:
362 const FloatMatrix &likes_;
364 float scale_;
365
366 public:
367 DecodableMatrixScaled(const DecodableMatrixScaled &) = delete;
368 DecodableMatrixScaled &operator=(const DecodableMatrixScaled &) = delete;
369};
