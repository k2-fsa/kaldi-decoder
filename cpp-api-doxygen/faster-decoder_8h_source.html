167 class Token {
168 public:
169 Arc arc_; // contains only the graph part of the cost;
170 // we can work out the acoustic part from difference between
171 // "cost_" and prev->cost_.
172 Token *prev_;
173 int32_t ref_count_ = 1;
174 // if you are looking for weight_ here, it was removed and now we just have
175 // cost_, which corresponds to ConvertToCost(weight_).
176 double cost_;
177 inline Token(const Arc &arc, float ac_cost, Token *prev)
178 : arc_(arc), prev_(prev) {
179 if (prev) {
180 prev->ref_count_++;
181 cost_ = prev->cost_ + arc.weight.Value() + ac_cost;
182 } else {
183 cost_ = arc.weight.Value() + ac_cost;
184 }
185 }
186
187 // for epsilon transitions, i.e., non-emitting arcs
188 inline Token(const Arc &arc, Token *prev) : arc_(arc), prev_(prev) {
189 if (prev) {
190 prev->ref_count_++;
191 cost_ = prev->cost_ + arc.weight.Value();
192 } else {
193 cost_ = arc.weight.Value();
194 }
195 }
196
197 inline bool operator<(const Token &other) const {
198 return cost_ > other.cost_;
199 }
200
201 inline static void TokenDelete(Token *tok) {
202 while (--tok->ref_count_ == 0) {
203 Token *prev = tok->prev_;
204 delete tok;
205 if (prev == nullptr) {
206 return;
207 } else {
208 tok = prev;
209 }
210 }
211 }
212 };
