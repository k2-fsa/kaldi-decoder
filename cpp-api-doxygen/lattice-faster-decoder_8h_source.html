136// We will template the decoder on the token type as well as the FST type; this
137// is a mechanism so that we can use the same underlying decoder code for
138// versions of the decoder that support quickly getting the best path
139// (LatticeFasterOnlineDecoder, see lattice-faster-online-decoder.h) and also
140// those that do not (LatticeFasterDecoder).

142// ForwardLinks are the links from a token to a token on the next frame.
143// or sometimes on the current frame (for input-epsilon links).
148template <typename Token>
149struct ForwardLink {
150 using Label = fst::StdArc::Label;
151
152 Token *next_tok; // the next token [or nullptr if represents final-state]
153 Label ilabel; // ilabel on arc
154 Label olabel; // olabel on arc
155 float graph_cost; // graph cost of traversing arc (contains LM, etc.)
156 float acoustic_cost; // acoustic cost (pre-scaled) of traversing arc
157 ForwardLink *next; // next in singly-linked list of forward arcs (arcs
158 // in the state-level lattice) from a token.
159 inline ForwardLink(Token *next_tok, Label ilabel, Label olabel,
160 float graph_cost, float acoustic_cost, ForwardLink *next)
161 : next_tok(next_tok),
162 ilabel(ilabel),
163 olabel(olabel),
164 graph_cost(graph_cost),
165 acoustic_cost(acoustic_cost),
166 next(next) {}
167};

171struct StdToken {
172 using ForwardLinkT = ForwardLink<StdToken>;
173 using Token = StdToken;
174
175 // Standard token type for LatticeFasterDecoder. Each active HCLG
176 // (decoding-graph) state on each frame has one token.
177
178 // tot_cost is the total (LM + acoustic) cost from the beginning of the
179 // utterance up to this point. (but see cost_offset_, which is subtracted
180 // to keep it in a good numerical range).
181 float tot_cost;
182
183 // exta_cost is >= 0. After calling PruneForwardLinks, this equals the
184 // minimum difference between the cost of the best path that this link is a
185 // part of, and the cost of the absolute best path, under the assumption that
186 // any of the currently active states at the decoding front may eventually
187 // succeed (e.g. if you were to take the currently active states one by one
188 // and compute this difference, and then take the minimum).
189 float extra_cost;
190
191 // 'links' is the head of singly-linked list of ForwardLinks, which is what we
192 // use for lattice generation.
193 ForwardLinkT *links;
194
195 // 'next' is the next in the singly-linked list of tokens for this frame.
196 Token *next;
197
198 // This function does nothing and should be optimized out; it's needed
199 // so we can share the regular LatticeFasterDecoderTpl code and the code
200 // for LatticeFasterOnlineDecoder that supports fast traceback.
201 inline void SetBackpointer(Token *backpointer) {}
202
203 // This constructor just ignores the 'backpointer' argument. That argument is
204 // needed so that we can use the same decoder code for LatticeFasterDecoderTpl
205 // and LatticeFasterOnlineDecoderTpl (which needs backpointers to support a
206 // fast way to obtain the best path).
207 inline StdToken(float tot_cost, float extra_cost, ForwardLinkT *links,
208 Token *next, Token *backpointer)
209 : tot_cost(tot_cost), extra_cost(extra_cost), links(links), next(next) {}
210};

214struct BackpointerToken {
215 using ForwardLinkT = ForwardLink<BackpointerToken>;
216 using Token = BackpointerToken;
217
218 // BackpointerToken is like Token but also
219 // Standard token type for LatticeFasterDecoder. Each active HCLG
220 // (decoding-graph) state on each frame has one token.
221
222 // tot_cost is the total (LM + acoustic) cost from the beginning of the
223 // utterance up to this point. (but see cost_offset_, which is subtracted
224 // to keep it in a good numerical range).
225 float tot_cost;
226
227 // exta_cost is >= 0. After calling PruneForwardLinks, this equals
228 // the minimum difference between the cost of the best path, and the cost of
229 // this is on, and the cost of the absolute best path, under the assumption
230 // that any of the currently active states at the decoding front may
231 // eventually succeed (e.g. if you were to take the currently active states
232 // one by one and compute this difference, and then take the minimum).
233 float extra_cost;
234
235 // 'links' is the head of singly-linked list of ForwardLinks, which is what we
236 // use for lattice generation.
237 ForwardLinkT *links;
238
239 // 'next' is the next in the singly-linked list of tokens for this frame.
240 BackpointerToken *next;
241
242 // Best preceding BackpointerToken (could be a on this frame, connected to
243 // this via an epsilon transition, or on a previous frame). This is only
244 // required for an efficient GetBestPath function in
245 // LatticeFasterOnlineDecoderTpl; it plays no part in the lattice generation
246 // (the "links" list is what stores the forward links, for that).
247 Token *backpointer;
248
249 inline void SetBackpointer(Token *backpointer) {
250 this->backpointer = backpointer;
251 }
252
253 inline BackpointerToken(float tot_cost, float extra_cost, ForwardLinkT *links,
254 Token *next, Token *backpointer)
255 : tot_cost(tot_cost),
256 extra_cost(extra_cost),
257 links(links),
258 next(next),
259 backpointer(backpointer) {}
260};

476 // prunes outgoing links for all tokens in active_toks_[frame]
477 // it's called by PruneActiveTokens
478 // all links, that have link_extra_cost > lattice_beam are pruned
479 // delta is the amount by which the extra_costs must change
480 // before we set *extra_costs_changed = true.
481 // If delta is larger, we'll tend to go back less far
482 // toward the beginning of the file.
483 // extra_costs_changed is set to true if extra_cost was changed for any token
484 // links_pruned is set to true if any link in any token was pruned
485 void PruneForwardLinks(int32_t frame_plus_one, bool *extra_costs_changed,
486 bool *links_pruned, float delta);
487
488 // This function computes the final-costs for tokens active on the final
489 // frame. It outputs to final-costs, if non-nullptr, a map from the Token*
490 // pointer to the final-prob of the corresponding state, for all Tokens
491 // that correspond to states that have final-probs. This map will be
492 // empty if there were no final-probs. It outputs to
493 // final_relative_cost, if non-nullptr, the difference between the best
494 // forward-cost including the final-prob cost, and the best forward-cost
495 // without including the final-prob cost (this will usually be positive), or
496 // infinity if there were no final-probs. [c.f. FinalRelativeCost(), which
497 // outputs this quanitity]. It outputs to final_best_cost, if
498 // non-nullptr, the lowest for any token t active on the final frame, of
499 // forward-cost[t] + final-cost[t], where final-cost[t] is the final-cost in
500 // the graph of the state corresponding to token t, or the best of
501 // forward-cost[t] if there were no final-probs active on the final frame.
502 // You cannot call this after FinalizeDecoding() has been called; in that
503 // case you should get the answer from class-member variables.
504 void ComputeFinalCosts(std::unordered_map<Token *, float> *final_costs,
505 float *final_relative_cost,

519 // Go backwards through still-alive tokens, pruning them if the
