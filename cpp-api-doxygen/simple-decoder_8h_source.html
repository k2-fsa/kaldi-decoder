126 fst::LatticeArc arc_; // We use LatticeArc so that we can separately

128 // we need to produce lattice-formatted output.
129 Token *prev_;
130 int32_t ref_count_ = 1;
131 double cost_; // accumulated total cost up to this point.
132 Token(const StdArc &arc, float acoustic_cost, Token *prev) : prev_(prev) {
133 arc_.ilabel = arc.ilabel;
134 arc_.olabel = arc.olabel;
135 arc_.weight = fst::LatticeWeight(arc.weight.Value(), acoustic_cost);
136 arc_.nextstate = arc.nextstate;
137 if (prev) {
138 prev->ref_count_++;
139 cost_ = prev->cost_ + (arc.weight.Value() + acoustic_cost);
140 } else {
141 cost_ = arc.weight.Value() + acoustic_cost;
142 }
143 }
144 bool operator<(const Token &other) { return cost_ > other.cost_; }
145
146 static void TokenDelete(Token *tok) {
147 while (--tok->ref_count_ == 0) {
148 Token *prev = tok->prev_;
149 delete tok;
150 if (prev == nullptr) {
151 return;
152 } else {
153 tok = prev;
154 }
155 }
156 if (tok->ref_count_ <= 0) {
157 throw std::runtime_error("SimpleDecoder::TokenDelete: ref_count_ <= 0");
158 }
159 }
160 };
