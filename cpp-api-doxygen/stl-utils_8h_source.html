21template <typename T>
22inline bool IsSorted(const std::vector<T> &vec) {
23 typename std::vector<T>::const_iterator iter = vec.begin(), end = vec.end();
24 if (iter == end) return true;
25 while (true) {
26 typename std::vector<T>::const_iterator next_iter = iter;
27 ++next_iter;
28 if (next_iter == end) return true; // end of loop and nothing out of order
29 if (*next_iter < *iter) return false;
30 iter = next_iter;
31 }
32}

56template <class T>
57inline void WriteIntegerVector(std::ostream &os, bool binary,
58 const std::vector<T> &v) {
59 // Compile time assertion that this is not called with a wrong type.
60 static_assert(std::is_integral_v<T>);
61 if (binary) {
62 char sz = sizeof(T); // this is currently just a check.
63 os.write(&sz, 1);
64 int32_t vecsz = static_cast<int32_t>(v.size());
65 if (static_cast<size_t>(vecsz) != v.size()) {
66 throw std::runtime_error("WriteIntegerVector: v.size() out of range");
67 }
68
69 os.write(reinterpret_cast<const char *>(&vecsz), sizeof(vecsz));
70 if (vecsz != 0) {
71 os.write(reinterpret_cast<const char *>(&(v[0])), sizeof(T) * vecsz);
72 }
73 } else {
74 // focus here is on prettiness of text form rather than
75 // efficiency of reading-in.
76 // reading-in is dominated by low-level operations anyway:
77 // for efficiency use binary.
78 os << "[ ";
79 typename std::vector<T>::const_iterator iter = v.begin(), end = v.end();
80 for (; iter != end; ++iter) {
81 if (sizeof(T) == 1)
82 os << static_cast<int16_t>(*iter) << " ";
83 else
84 os << *iter << " ";
85 }
86 os << "]\n";
87 }
88 if (os.fail()) {
89 throw std::runtime_error("Write failure in WriteIntegerVector.");
90 }
91}

93template <class T>
94inline void ReadIntegerVector(std::istream &is, bool binary,
95 std::vector<T> *v) {
96 static_assert(std::is_integral_v<T>);
97 if (v == nullptr) {
98 throw std::runtime_error("ReadIntegerVector: v != nullptr");
99 }
100 if (binary) {
101 int sz = is.peek();
102 if (sz == sizeof(T)) {
103 is.get();
104 } else { // this is currently just a check.
105 std::ostringstream os;
106 os << "ReadIntegerVector: expected to see type of size " << sizeof(T)
107 << ", saw instead " << sz << ", at file position " << is.tellg();
108 throw std::runtime_error(os.str());
109 }
110 int32_t vecsz = 0;
111 is.read(reinterpret_cast<char *>(&vecsz), sizeof(vecsz));
112 if (is.fail() || vecsz < 0) {
113 goto bad; // NOLINT -- verbatim kaldi src/base/io-funcs-inl.h
114 }
115
116 v->resize(vecsz);
117
118 if (vecsz > 0) {
119 is.read(reinterpret_cast<char *>(&((*v)[0])), sizeof(T) * vecsz);
120 }
121 } else {
122 std::vector<T> tmp_v; // use temporary so v doesn't use extra memory
123 // due to resizing.
124 is >> std::ws;
125 if (is.peek() != static_cast<int>('[')) {
126 std::ostringstream os;
127 os << "ReadIntegerVector: expected to see [, saw " << is.peek()
128 << ", at file position " << is.tellg();
129 throw std::runtime_error(os.str());
130 }
131 is.get(); // consume the '['.
132 is >> std::ws; // consume whitespace.
133 while (is.peek() != static_cast<int>(']')) {
134 if (sizeof(T) == 1) { // read/write chars as numbers.
135 int16_t next_t = 0;
136 is >> next_t >> std::ws;
137 if (is.fail()) {
138 goto bad; // NOLINT -- verbatim kaldi src/base/io-funcs-inl.h
139 } else {
140 tmp_v.push_back(static_cast<T>(next_t));
141 }
142 } else {
143 T next_t;
144 is >> next_t >> std::ws;
145 if (is.fail()) {
146 goto bad; // NOLINT -- verbatim kaldi src/base/io-funcs-inl.h
147 } else {
148 tmp_v.push_back(next_t);
149 }
150 }
151 }
152 is.get(); // get the final ']'.
153 *v = tmp_v; // could use std::swap to use less temporary memory, but this
154 // uses less permanent memory.
155 }
156 if (!is.fail()) return;
157bad: {
158 std::ostringstream os;
159 os << "ReadIntegerVector: read failure at file position " << is.tellg();
160 throw std::runtime_error(os.str());
161}
162}

166template <class A>
167void DeletePointers(std::vector<A *> *v) {
168 if (v == nullptr) {
169 throw std::runtime_error("DeletePointers: v != nullptr");
170 }
171 typename std::vector<A *>::iterator iter = v->begin(), end = v->end();
172 for (; iter != end; ++iter) {
173 if (*iter != nullptr) {
174 delete *iter;
175 *iter = nullptr; // set to nullptr for extra safety.
176 }
177 }
178}

181template <typename Int1, typename Int2 = Int1>
182struct PairHasher { // hashing function for pair<int>
183 size_t operator()(const std::pair<Int1, Int2> &x) const noexcept {
184 // 7853 was chosen at random from a list of primes.
185 return x.first + x.second * 7853;
186 }
187 PairHasher() { // Check we're instantiated with an integer type.
188 static_assert(std::is_integral_v<Int1>);
189 static_assert(std::is_integral_v<Int2>);
190 }
191};

194template <class A>
195bool ContainsNullPointers(const std::vector<A *> &v) {
196 typename std::vector<A *>::const_iterator iter = v.begin(), end = v.end();
197 for (; iter != end; ++iter)
198 if (*iter == static_cast<A *>(nullptr)) return true;
199 return false;
200}

203template <typename Int>
204struct VectorHasher { // hashing function for vector<Int>.
205 size_t operator()(const std::vector<Int> &x) const noexcept {
206 size_t ans = 0;
207 auto iter = x.begin(), end = x.end();
208 for (; iter != end; ++iter) {
209 ans *= kPrime;
210 ans += *iter;
211 }
212 return ans;
213 }
214 VectorHasher() { // Check we're instantiated with an integer type.
215 static_assert(std::is_integral_v<Int>);
216 }
217
218 private:
219 static const int kPrime = 7853;
220};
